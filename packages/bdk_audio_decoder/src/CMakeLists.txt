cmake_minimum_required(VERSION 3.14)
project(bdk_audio_decoder LANGUAGES CXX)
add_library(bdk_audio_decoder SHARED bdk_audio_decoder.cpp)
if(WIN32)
  target_sources(bdk_audio_decoder PRIVATE bdk_audio_windows.cpp)
  target_link_libraries(bdk_audio_decoder PRIVATE mfplat mfreadwrite mfuuid ole32 propsys)
  if(MSVC)
    target_link_libraries(bdk_audio_decoder PRIVATE delayimp)
    target_link_options(bdk_audio_decoder PRIVATE "/DELAYLOAD:MFPlat.DLL" "/DELAYLOAD:MFReadWrite.dll")
  endif()
elseif(ANDROID)
  target_sources(bdk_audio_decoder PRIVATE bdk_audio_android.cpp)
  target_link_libraries(bdk_audio_decoder PRIVATE mediandk)
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(bdk_audio_decoder PRIVATE bdk_audio_apple.mm)
  set_source_files_properties(bdk_audio_apple.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(bdk_audio_decoder PRIVATE "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreMedia" "-framework Foundation")
else()
  target_sources(bdk_audio_decoder PRIVATE bdk_audio_unsupported.cpp)
endif()
option(BDK_DECODER_BUILD_TESTS "Build hardware-free PCM window tests" OFF)
option(BDK_DECODER_BUILD_CLI "Build the offline C ABI diagnostic CLI" OFF)
option(BDK_DECODER_DIAGNOSTICS "Enable test-only native decoder tracing" OFF)
if(BDK_DECODER_DIAGNOSTICS)
  target_compile_definitions(bdk_audio_decoder PRIVATE BDK_DECODER_DIAGNOSTICS=1)
endif()
if(BDK_DECODER_BUILD_CLI)
  add_executable(bdk_decoder_qa ../native_test/decoder_qa.cpp)
  target_compile_features(bdk_decoder_qa PRIVATE cxx_std_17)
  target_include_directories(bdk_decoder_qa PRIVATE .)
  target_link_libraries(bdk_decoder_qa PRIVATE bdk_audio_decoder)
  if(ANDROID)
    target_link_libraries(bdk_decoder_qa PRIVATE dl)
  endif()
endif()
if(BDK_DECODER_BUILD_TESTS AND NOT ANDROID AND NOT APPLE)
  add_executable(bdk_pcm_window_test ../native_test/pcm_window_test.cpp bdk_audio_decoder.cpp bdk_audio_unsupported.cpp)
  target_compile_features(bdk_pcm_window_test PRIVATE cxx_std_17)
  target_include_directories(bdk_pcm_window_test PRIVATE .)
  enable_testing()
  add_test(NAME pcm_window COMMAND bdk_pcm_window_test)
endif()
target_compile_features(bdk_audio_decoder PRIVATE cxx_std_17)
set_target_properties(bdk_audio_decoder PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(bdk_audio_decoder PRIVATE /W3 /fp:precise)
else()
  target_compile_options(bdk_audio_decoder PRIVATE -fno-fast-math)
  target_link_libraries(bdk_audio_decoder PRIVATE m)
endif()
if(ANDROID)
  target_link_options(bdk_audio_decoder PRIVATE "-Wl,-z,max-page-size=16384")
endif()
